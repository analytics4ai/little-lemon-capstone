1. RELATIONAL DATABASE SCHEMA & ER DIAGRAM STRUCTURE (3NF)
================================================================================

Entities & Relationships:
- Customers (1) ----< (Many) Bookings
- Customers (1) ----< (Many) Orders
- MenuItems (1) ----< (Many) Orders
- Orders    (1) ----< (Many) OrderDeliveryStatus

Table Specifications:
1. Customers: CustomerID (PK), CustomerName, City, Country, PostalCode, CountryCode
2. MenuItems: MenuItemID (PK), CourseName, CuisineName, StarterName, DesertName, Drink, Sides
3. Bookings:  BookingID (PK), CustomerID (FK), BookingDate, TableNumber
4. Orders:    OrderID (PK), OrderDate, CustomerID (FK), MenuItemID (FK), Quantity, Cost, Sales, Discount
5. OrderDeliveryStatus: DeliveryID (PK), OrderID (FK), DeliveryDate, DeliveryCost, Status

DROP DATABASE IF EXISTS LittleLemonDB;
CREATE DATABASE LittleLemonDB;
USE LittleLemonDB;

-- 1. Customers Table
CREATE TABLE Customers (
    CustomerID VARCHAR(20) PRIMARY KEY,
    CustomerName VARCHAR(100) NOT NULL,
    City VARCHAR(100),
    Country VARCHAR(100),
    PostalCode VARCHAR(20),
    CountryCode VARCHAR(10)
);

-- 2. MenuItems Table
CREATE TABLE MenuItems (
    MenuItemID INT AUTO_INCREMENT PRIMARY KEY,
    CourseName VARCHAR(100),
    CuisineName VARCHAR(100),
    StarterName VARCHAR(100),
    DesertName VARCHAR(100),
    Drink VARCHAR(100),
    Sides VARCHAR(100)
);

-- 3. Bookings Table
CREATE TABLE Bookings (
    BookingID INT AUTO_INCREMENT PRIMARY KEY,
    CustomerID VARCHAR(20) NOT NULL,
    BookingDate DATE NOT NULL,
    TableNumber INT NOT NULL,
    FOREIGN KEY (CustomerID) REFERENCES Customers(CustomerID) ON DELETE CASCADE
);

-- 4. Orders Table
CREATE TABLE Orders (
    OrderID VARCHAR(20) PRIMARY KEY,
    OrderDate DATE NOT NULL,
    CustomerID VARCHAR(20) NOT NULL,
    MenuItemID INT NOT NULL,
    Quantity INT NOT NULL,
    Cost DECIMAL(10, 2) NOT NULL,
    Sales DECIMAL(10, 2) NOT NULL,
    Discount DECIMAL(10, 2) DEFAULT 0.00,
    FOREIGN KEY (CustomerID) REFERENCES Customers(CustomerID) ON DELETE CASCADE,
    FOREIGN KEY (MenuItemID) REFERENCES MenuItems(MenuItemID) ON DELETE CASCADE
);

-- 5. OrderDeliveryStatus Table
CREATE TABLE OrderDeliveryStatus (
    DeliveryID INT AUTO_INCREMENT PRIMARY KEY,
    OrderID VARCHAR(20) NOT NULL,
    DeliveryDate DATE,
    DeliveryCost DECIMAL(10, 2) DEFAULT 0.00,
    Status VARCHAR(50) DEFAULT 'Pending',
    FOREIGN KEY (OrderID) REFERENCES Orders(OrderID) ON DELETE CASCADE
);

-- REQUIRED STORED PROCEDURES
-- =============================================================================

-- 1. GetMaxQuantity()
DELIMITER //
CREATE PROCEDURE GetMaxQuantity()
BEGIN
    SELECT MAX(Quantity) AS "Max Quantity in Order" FROM Orders;
END //
DELIMITER ;

-- 2. ManageBooking()
DELIMITER //
CREATE PROCEDURE ManageBooking(IN booking_date DATE, IN table_no INT)
BEGIN
    DECLARE booking_status INT DEFAULT 0;
    SELECT COUNT(*) INTO booking_status
    FROM Bookings
    WHERE BookingDate = booking_date AND TableNumber = table_no;
    
    IF booking_status > 0 THEN
        SELECT CONCAT("Table ", table_no, " is already booked - booking cancelled") AS "Booking Status";
    ELSE
        SELECT CONCAT("Table ", table_no, " is available") AS "Booking Status";
    END IF;
END //
DELIMITER ;

-- 3. AddBooking()
DELIMITER //
CREATE PROCEDURE AddBooking(IN booking_id INT, IN customer_id VARCHAR(20), IN booking_date DATE, IN table_no INT)
BEGIN
    INSERT INTO Bookings (BookingID, CustomerID, BookingDate, TableNumber)
    VALUES (booking_id, customer_id, booking_date, table_no);
    SELECT "New booking added" AS Confirmation;
END //
DELIMITER ;

-- 4. UpdateBooking()
DELIMITER //
CREATE PROCEDURE UpdateBooking(IN booking_id INT, IN booking_date DATE)
BEGIN
    UPDATE Bookings
    SET BookingDate = booking_date
    WHERE BookingID = booking_id;
    SELECT CONCAT("Booking ", booking_id, " updated") AS Confirmation;
END //
DELIMITER ;

-- 5. CancelBooking()
DELIMITER //
CREATE PROCEDURE CancelBooking(IN booking_id INT)
BEGIN
    DELETE FROM Bookings
    WHERE BookingID = booking_id;
    SELECT CONCAT("Booking ", booking_id, " cancelled") AS Confirmation;
END //
DELIMITER ;

import mysql.connector as connector

def get_connection():
    """Establish connection to the LittleLemonDB MySQL database."""
    return connector.connect(
        user="root",
        password="your_password",
        host="127.0.0.1",
        database="LittleLemonDB"
    )

def main():
    try:
        conn = get_connection()
        cursor = conn.cursor()
        print("Connected to LittleLemonDB successfully.\n")

        # Show all tables in the database
        cursor.execute("SHOW TABLES;")
        print("Tables in LittleLemonDB:")
        for table in cursor.fetchall():
            print(f" - {table[0]}")

        # Query customers with orders exceeding $60
        query_high_value = """
        SELECT c.CustomerName, c.City, c.Country, o.OrderID, o.Sales
        FROM Customers c
        INNER JOIN Orders o ON c.CustomerID = o.CustomerID
        WHERE o.Sales > 60;
        """
        cursor.execute(query_high_value)
        print("\nCustomers with Sales > $60:")
        for row in cursor.fetchall():
            print(row)

        # Call Stored Procedures
        print("\n--- Testing Stored Procedures ---")
        
        cursor.callproc("GetMaxQuantity")
        for result in cursor.stored_results():
            print("GetMaxQuantity:", result.fetchall())

        cursor.callproc("ManageBooking", ("2022-10-10", 5))
        for result in cursor.stored_results():
            print("ManageBooking:", result.fetchall())

        cursor.callproc("AddBooking", (4, "72-055-7985", "2022-12-15", 6))
        conn.commit()
        for result in cursor.stored_results():
            print("AddBooking:", result.fetchall())

        cursor.callproc("UpdateBooking", (4, "2022-12-20"))
        conn.commit()
        for result in cursor.stored_results():
            print("UpdateBooking:", result.fetchall())

        cursor.callproc("CancelBooking", (4,))
        conn.commit()
        for result in cursor.stored_results():
            print("CancelBooking:", result.fetchall())

    except connector.Error as err:
        print(f"Database Error: {err}")
    finally:
        if 'conn' in locals() and conn.is_connected():
            cursor.close()
            conn.close()
            print("\nMySQL connection closed.")

if __name__ == "__main__":
    main()
